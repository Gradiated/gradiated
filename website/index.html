import { Gradiated } from "@gradiated/sdk"

const run = await gradiated.runs.create({
  task: "Research and compare three vendors",
  tools: [webSearch, browser],
  profile: {
    budget: { max_usd: 2 }, deadline: "20m", quality: "thorough"
  }
})
